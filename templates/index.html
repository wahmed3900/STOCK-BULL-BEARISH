import os import json import logging from typing import Generator from openai import OpenAI from tenacity import retry, stop_after_attempt, wait_exponential logging.basicConfig(level=logging.INFO) logger = logging.getLogger(__name__) class OpenRouterStreamer: def __init__(self, api_key: str = None): self.api_key = api_key or os.getenv("OPENROUTER_API_KEY") if not self.api_key: raise ValueError("OpenRouter API key is required") self.client = OpenAI( api_key=self.api_key, base_url="https://openrouter.ai/api/v1", timeout=30.0 ) @retry(stop=stop_after_attempt(3), wait=wait_exponential(multiplier=1, min=2, max=10)) def stream_analysis(self, symbol: str, news: str, model: str = "deepseek-ai/DeepSeek-R1") -> Generator[str, None, None]: try: # Yield start metadata yield f"data: {json.dumps({'type': 'start', 'symbol': symbol, 'model': model})}\n\n" stream = self.client.chat.completions.create( model=model, messages=[{ "role": "user", "content": f"Analyze {symbol} using this news:\n{news}\nGive step-by-step reasoning." }], stream=True ) for chunk in stream: if chunk.choices: token = chunk.choices[0].delta.get("content", "") if token: yield f"data: {json.dumps({'type': 'token', 'content': token})}\n\n" # Yield end metadata yield f"data: {json.dumps({'type': 'end'})}\n\n" except Exception as e: logger.error(f"Streaming error for {symbol}: {e}", exc_info=True) yield f"data: {json.dumps({'type': 'error', 'message': str(e)})}\n\n"
